type Fn = (event: { nativeEvent: unknown }) => unknown;
const host = (root: ReturnType<typeof createRoot>) =>
  root.container.queryAll((node) => node.type === 'MinimalWebView')[0]?.props as Record<
    string,
    Fn & string
  >;
async function mount(props: Partial<WebViewProps>) {
  const root = createRoot();
  await act(async () =>
    root.render(<WebView source={{ uri: 'https://example.com' }} {...props} />)
  );
  return {
    root,
    native: () => host(root),
    fire: (name: string, nativeEvent: unknown) =>
      act(async () => host(root)[name]!({ nativeEvent })),
  };
}
const state = (loading: boolean) => ({
  url: 'https://example.com/',
  pendingUrl: null,
  title: '',
  loading,
  progress: 0,
  canGoBack: false,
  canGoForward: false,
  outcome: null,
});

describe('a WebView with none of the optional callbacks', () => {
  it('takes every native event without throwing or logging an error', async () => {
    const error = jest.spyOn(console, 'error').mockImplementation(() => {});
    const { fire } = await mount({});
    await fire('onNavigationState', state(true));
    await fire('onNavigationState', state(false));
    await fire('onNavigation', { url: 'https://a/', mainFrame: true, action: 'external' });
    await fire('onProcessGone', { didCrash: true, recovered: true });
    await fire('onError', { kind: 'http', url: 'https://a/', code: 500, description: '' });
    await fire('onBridge', { message: '{}', origin: 'https://a' });
    await fire('onBridge', { navigated: true });
    await fire('onBridge', { unavailable: true });
    await fire('onHookEvent', { name: 'hook.note', payload: {} });
    await fire('onHookEvent', { name: 'hook.ask', payload: {}, askId: 1 });
    await fire('onHookEvent', { name: 'core.fileRequest', payload: {}, askId: 2 });
    expect(error).not.toHaveBeenCalled();
    expect(mockAnswer.mock.calls).toEqual([
      [1, { value: null }],
      [2, { value: null }],
    ]);
  });
});

describe('the bridge glue', () => {
  const setup = async () => {
    const bridge = createBridge({ allowedOrigins: ['https://example.com'], timeoutMs: 1000 });
    const view = await mount({ bridge });
    return { bridge, ...view };
  };
  const ready = JSON.stringify({ doc: 'd1', kind: 'ready' });

  it('hands native the page client and the origins, and posts what the bridge sends', async () => {
    const { bridge, native, fire } = await setup();
    expect(native().bridgeScript).toContain('nativeBridge');
    expect(native().bridgeOrigins).toEqual(['https://example.com']);
    bridge.emit('hello', 1);
    expect(mockBridgePost).not.toHaveBeenCalled();
    await fire('onBridge', { message: ready, origin: 'https://example.com' });
    expect(mockBridgePost).toHaveBeenCalledTimes(1);
  });

  it("gives a page's call the origin native verified", async () => {
    const { bridge, fire } = await setup();
    const senders: unknown[] = [];
    bridge.handle('who', (_, sender) => senders.push(sender));
    await fire('onBridge', {
      message: JSON.stringify({ id: 'p1', doc: 'd1', kind: 'call', method: 'who' }),
      origin: 'https://example.com:8443',
    });
    expect(senders).toEqual([{ origin: 'https://example.com:8443' }]);
  });

  it('rejects calls with navigated or unavailable as native reports', async () => {
    const { bridge, fire } = await setup();
    await fire('onBridge', { message: ready, origin: 'https://example.com' });
    const inFlight = bridge.call('x');
    await fire('onBridge', { navigated: true });
    await expect(inFlight).rejects.toMatchObject({ code: 'navigated' });
    await fire('onBridge', { unavailable: true });
    await expect(bridge.call('x')).rejects.toMatchObject({ code: 'unavailable' });
  });

  it('detaches at unmount, so the bridge can attach to the next WebView', async () => {
    const { bridge, root } = await setup();
    await act(async () => root.unmount());
    const next = createRoot();
    await act(async () =>
      next.render(<WebView source={{ uri: 'https://example.com' }} bridge={bridge} />)
    );
  });
});

describe('the creation key', () => {
  const remounts = async (...shows: Partial<WebViewProps>[]) => {
    const root = createRoot();
    mockMount.mockClear();
    for (const props of shows)
      await act(async () =>
        root.render(<WebView source={{ uri: 'https://example.com' }} {...props} />)
      );
    const mounts = mockMount.mock.calls.length;
    await act(async () => root.unmount());
    return mounts;
  };
  const bridge = (): Bridge => createBridge({ allowedOrigins: ['https://a.example'] });

  it('recreates the WebView for another bridge, and keeps it for the same one', async () => {
    const a = bridge();
    expect(await remounts({ bridge: a }, { bridge: a })).toBe(1);
    expect(await remounts({ bridge: bridge() }, { bridge: bridge() })).toBe(2);
    expect(await remounts({}, { bridge: bridge() })).toBe(2);
  });

  it('keeps the WebView when ios.configuration only changes its key order', async () => {
    const one = { configuration: { x: 1, nested: { p: 1, q: 2 } } } as never;
    const two = { configuration: { nested: { q: 2, p: 1 }, x: 1 } } as never;
    expect(await remounts({ ios: one }, { ios: two })).toBe(1);
    expect(await remounts({ ios: one }, { ios: { configuration: { x: 2 } } as never })).toBe(2);
  });
});

    const ref = createRef<WebViewRef>();
    const root = createRoot();
    await act(async () =>
      root.render(<WebView ref={ref} source={{ uri: 'https://example.com' }} />)
    );
    const handle = ref.current!;
    mockNavigate.mockClear();
    handle.goBack();
    handle.goForward();
    handle.reload();
    handle.stopLoading();

    await act(async () => root.unmount());

      handle.goBack();
      handle.goForward();
      handle.reload();
      handle.stopLoading();
    }).not.toThrow();
  });
});

describe('asks in flight', () => {
  it('are aborted at unmount, and a finished ask is no longer cancellable', async () => {
    const signals: AbortSignal[] = [];
    const { fire, root, native } = await mount({
      onPermissionRequest: (_, signal) => {
        signals.push(signal);
        return signal.aborted ? false : new Promise<boolean>(() => {});
      },
      onFileRequest: (_, signal) => {
        signals.push(signal);
        return [];
      },
    });
    // Not awaited: this ask stays open until the unmount.
    native()['onHookEvent']!({
      nativeEvent: { name: 'core.permissionRequest', payload: {}, askId: 1 },
    });
    await fire('onHookEvent', { name: 'core.fileRequest', payload: {}, askId: 2 });
    await fire('onHookEvent', { name: 'core.askCancelled', payload: { askId: 2 } });
    expect(signals.map((s) => s.aborted)).toEqual([false, false]);
    await act(async () => root.unmount());
    expect(signals.map((s) => s.aborted)).toEqual([true, false]);
  });
});

describe('the ref objects', () => {
  it('unsubscribe stops a listener, and a ref swap hands the new ref the current state', async () => {
    const { createWebViewRef } = jest.requireActual<typeof import('../WebView')>('../WebView');
    const one = createWebViewRef();
    const two = createWebViewRef();
    const listener = jest.fn();
    const unsubscribe = one.subscribe(listener);
    const root = createRoot();
    const show = (ref: typeof one) =>
      act(async () => root.render(<WebView ref={ref} source={{ uri: 'https://example.com' }} />));
    await show(one);
    listener.mockClear();
    unsubscribe();
    await act(async () => host(root)['onNavigationState']!({ nativeEvent: state(true) }));
    expect(listener).not.toHaveBeenCalled();
    await show(two);
    expect(two.getState().loading).toBe(true);
    expect(two.current).not.toBeNull();
  });
});
